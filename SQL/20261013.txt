DROP table t_board;
DROP TABLE t_user;

CREATE TABLE t_user(
id SERIAL PRIMARY KEY,
username VARCHAR,
password VARCHAR(2000),
phoneno VARCHAR,
gender VARCHAR CHECK(gender IN('m','f')) DEFAULT 'm',
created_dt TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE t_board(
id SERIAL PRIMARY KEY,
title VARCHAR,
content TEXT,
user_id INT4 NOT NULL,

CONSTRAINT fk_board_user
FOREIGN KEY(user_id)
REFERENCES t_user(id)
ON UPDATE CASCADE
ON DELETE CASCADE,

created_dt TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE t_comment(
id SERIAL PRIMARY KEY,
board_id INT4 NOT NULL,
user_id INT4 NOT NULL,
cmt VARCHAR,
created_dt TIMESTAMPTZ DEFAULT NOW(),

CONSTRAINT fk_cmt_board
FOREIGN KEY(board_id)
REFERENCES t_board(id)
ON UPDATE CASCADE
ON DELETE CASCADE,

CONSTRAINT fk_cmt_user
FOREIGN KEY(user_id)
REFERENCES t_user(id)
ON UPDATE CASCADE
ON DELETE CASCADE
);
