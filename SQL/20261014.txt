ALTER TABLE t_board
ADD COLUMN viewcnt INT4 DEFAULT 0 CHECK( viewcnt>=0);

ALTER TABLE t_board
ADD COLUMN likecnt INT4 DEFAULT 0 CHECK( likecnt>=0);


INSERT INTO t_board(title, content, user_id, viewcnt, likecnt)
SELECT
    substring(md5(random()::text) FROM 1 FOR 10),
    substring(md5(random()::text) FROM 1 FOR 10),
    (SELECT id FROM t_user ORDER BY random() LIMIT 1),
    floor(random() * 45 + 1)::int,
    floor(random() * 45 + 1)::int
FROM generate_series(1, 100);

INSERT INTO t_comment(board_id, user_id, cmt)
SELECT
    (SELECT id FROM t_board ORDER BY random() LIMIT 1),
    (SELECT id FROM t_user ORDER BY random() LIMIT 1),
    substring(md5(random()::text) FROM 1 FOR 10)
FROM generate_series(1, 100);


SELECT
b.id AS "board_id"
,b.user_id
,b.title
,b.content
,b.viewcnt
,b.likecnt
,u.username
,b.created_dt
FROM t_board AS b
JOIN t_user AS u ON b.user_id = u.id
ORDER BY created_dt DESC
LIMIT 1000
;

CREATE INDEX idx_board_created_dt
ON t_board(created_dt);

SELECT
b.user_id
,MAX(u.username) AS "username"
,COUNT(b.id) AS "작성한개시글수"
FROM t_board as b
JOIN t_user as u ON b.user_id=u.id
GROUP BY b.user_id
;
